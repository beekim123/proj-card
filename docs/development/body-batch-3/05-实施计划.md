```gdscript
func debug_deal_roundtrip(tree: SceneTree, directory: String) -> bool:
    var old_dir: String = OS.get_environment("GEMCARD_SAVE_DIR")
    OS.set_environment("GEMCARD_SAVE_DIR", directory)
    var main: GDScript = load("res://scripts/app/game.gd")
    var run: GDScript = load("res://rules/run.gd")
    var app: Control = main.new()
    app.debug_enabled = true
    tree.root.add_child(app)
    await tree.process_frame
    app.start_adventure()
    run.choose_relic(app.run_state, 0)
    app.debug_deal([1, "potion", "palette", "diamond"])
    var ok: bool = app.storage.last_error.is_empty()
    var loaded: Dictionary = app.storage.load_run()
    ok = ok and loaded.get("state") is Dictionary
    if loaded.get("state") is Dictionary:
        var state: Dictionary = loaded.state
        ok = ok and state.cards.size() == 33 and int(state.nextCardId) == 34
        ok = ok and state.game.hand.size() == 4
        if state.game.hand.size() == 4:
            ok = ok and int(state.game.hand[0].v) == 1
            ok = ok and state.game.hand[1].get("type") == "potion"
            ok = ok and state.game.hand[2].get("type") == "palette"
            ok = ok and state.game.hand[3].get("type") == "diamond"
    await app.shutdown_presentation()
    app.queue_free()
    await tree.process_frame
    OS.set_environment("GEMCARD_SAVE_DIR", old_dir)
    return ok
```

```gdscript
extends RefCounted
const R: GDScript = preload("res://rules/run.gd")
const RC: GDScript = preload("res://rules/run_config.gd")
static func full_victory(tier: int = 0) -> Dictionary:
    var state: Dictionary = R.create_run(1979, {
        "tier":tier,"unlockedTier":5,"tuningScale":1,
        "runId":"batch3-staged-tier-"+str(tier)})
    R.choose_relic(state, 0)
    R.debug_jump_act_room(state, 3, 6)
    for id: String in RC.RELIC_IDS:
        if state.relics.size() >= 6: break
        R.acquire_relic(state, id)
    R.debug_set_monster_hp(state, 0)
    R.end_run_turn(state)
    return state
```

done
